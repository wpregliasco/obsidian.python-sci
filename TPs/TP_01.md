```python
# Tiempos
times = []
for row in range(1, len(data_lines)):
    if data_lines[row] != '':
        times.append(data_lines[row].split(',')[1])


for item in times[:10]:
    print(item)

```

```python
# Metar
metar = []
for row in range(1, len(data_lines)):
    if data_lines[row] != '':
        metar.append(data_lines[row].split(',')[5])

for item in metar[:10]:
    print(item)
```

```python
# Persión atmosférica
press = []
for row in metar:
    found = False
    for item in row.split():
        if item.startswith('Q'):
            press.append(item)
            found = True
    if found == False:
        press.append(None)

for item in press[:10]:
    print(item)            
```
